    <script>
        // 🔒 Manual Lockdown Handler
        function lockdownConsoleManual() {
            console.warn("SECURITY OVERRIDE: Manual session termination initialized.");
            sessionStorage.clear();
            localStorage.clear();
            window.location.href = "index.html";
        }

        // 🛰️ CORE VIEW ENGINE: Pulls live records from Flask -> SQL Server Express
        async function loadDashboardTickets() {
            try {
                const selectedTenant = document.getElementById('tenantFilter')?.value;

                let url = '/api/dashboard/tickets';
                if (selectedTenant && selectedTenant !== 'ALL') {
                    url += `?client=${selectedTenant}`;
                }

                console.log(`📡 Requesting isolated payload from: ${url}`);
                const response = await fetch(url);
                if (!response.ok) throw new Error(`HTTP network error: ${response.status}`);
                const tickets = await response.json();

                renderDashboardCards(tickets);

            } catch (error) {
                console.warn("⚠️ Flask API offline. Loading fallback telemetry into console grid...", error);

                const mockTickets = [
                    { id: 101, deviceName: "DC-01 (Active Directory)", alertType: "Kerberos TGT Anomaly", tenantId: "ALPHA_CORP", status: "NEW", severity: "HIGH", riskScore: 88, timestamp: "2026-08-04 16:35", assignmentGroup: "SecOps Triage", notes: "Suspicious ticket requests detected." },
                    { id: 102, deviceName: "SQL-DB-PROD", alertType: "Failed Auth Threshold Exceeded", tenantId: "BETA_INC", status: "ASSIGNED", severity: "CRITICAL", riskScore: 94, timestamp: "2026-08-04 16:12", assignmentGroup: "Database Sec", notes: "Investigating IP 192.168.1.105." },
                    { id: 103, deviceName: "FW-EDGE-02", alertType: "Port Scan Mitigated", tenantId: "ALPHA_CORP", status: "RESOLVED", severity: "LOW", riskScore: 12, timestamp: "2026-08-04 15:45", assignmentGroup: "NetOps", notes: "Automated drop rule applied." }
                ];

                renderDashboardCards(mockTickets);
            }
        }

        // Render expandable ticket cards
        function renderDashboardCards(tickets) {
            const container = document.getElementById('ticketContainer');
            if (!container) return;

            container.innerHTML = '';

            if (!tickets || tickets.length === 0) {
                container.innerHTML = '<p style="color: #94a3b8; font-family: monospace;">No active records logged in database query.</p>';
                return;
            }

            tickets.forEach(ticket => {
                const card = document.createElement('div');
                card.className = 'ticket-card';
                card.style.display = 'block';
                card.setAttribute('data-tenant', ticket.tenantId || 'ALL');

                card.innerHTML = `
                    <details style="width: 100%; cursor: pointer;">
                        <summary style="display: flex; justify-content: space-between; align-items: center; list-style: none;">
                            <div style="display: flex; gap: 15px; font-family: monospace;">
                                <span style="color: #00ff66; font-weight: bold;">🆔 #${ticket.id}</span>
                                <strong style="color: #38bdf8;">${ticket.deviceName || 'Unknown'}</strong>
                                <span style="color: #94a3b8;">[${ticket.alertType || 'Alert'}]</span>
                            </div>
                            <div style="display: flex; gap: 10px; align-items: center; font-family: monospace;">
                                <span style="color: #a855f7; font-weight: bold; font-size: 12px; background: #1e1b4b; padding: 2px 6px; border-radius: 3px;">${ticket.tenantId || 'GLOBAL'}</span>
                                <span style="color: ${ticket.status === 'NEW' ? '#10b981' : '#f59e0b'}; font-size: 12px;">● ${ticket.status || 'NEW'}</span>
                                <span style="color: #64748b; font-size: 14px;">▼</span>

        // 💾 Save Journal Notes with Auto-Timestamping
        function saveTicketLifecycle(ticketId) {
            const textarea = document.getElementById(`notes-${ticketId}`);
            if (!textarea) return;

            const rawNotes = textarea.value.trim();
            if (!rawNotes) return;

            const now = new Date();
            const YYYY = now.getFullYear();
            const MM = String(now.getMonth() + 1).padStart(2, '0');
            const DD = String(now.getDate()).padStart(2, '0');
            const hh = String(now.getHours()).padStart(2, '0');
            const mm = String(now.getMinutes()).padStart(2, '0');
            const currentTimestamp = `[${YYYY}-${MM}-${DD} ${hh}:${mm}]`;

            const lines = rawNotes.split('\n');
            let formattedText = '';
            let inUntaggedBlock = false;

            lines.forEach((line, index) => {
                const trimmed = line.trim();
                const hasHeader = /^\[\d{4}-\d{2}-\d{2}/.test(trimmed);

                if (hasHeader) {
                    inUntaggedBlock = false;
                    formattedText += (index === 0 ? '' : '\n\n') + trimmed;
                } else if (trimmed === '') {
                    inUntaggedBlock = false;
                } else {
                    if (!inUntaggedBlock && !formattedText.endsWith(currentTimestamp)) {
                        formattedText += (formattedText ? '\n\n' : '') + `${currentTimestamp}\n${trimmed}`;
                        inUntaggedBlock = true;
                    } else {
                        formattedText += `\n${trimmed}`;
                    }
                }
            });

            textarea.value = formattedText;

            fetch(`/api/tickets/${ticketId}/notes`, {
                method: 'PUT',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ notes: formattedText })
            })
                .then(response => {
                    if (!response.ok) throw new Error(`HTTP status ${response.status}`);
                    return response.json();
                })
                .then(data => alert(`Journal entry committed for Ticket #${ticketId}!`))
                .catch(error => alert('Failed to save log entry.'));
        }

        // Send Status Changes to Backend
        async function handleStatusChange(ticketId, newStatus) {
            try {
                const response = await fetch(`/api/tickets/${ticketId}`, {
                    method: 'PUT',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ status: newStatus })
                });

                if (response.ok) {
                    console.log(`[TICKET UPDATE] Ticket #${ticketId} changed to ${newStatus}`);
                    loadDashboardTickets();
                } else {
                    const result = await response.json();
                    alert(`Failed to update status: ${result.message}`);
                }
            } catch (err) {
                console.error('[NETWORK ERROR]', err);
            }
        }

        // 🛠️ Local Diagnostic Tool Launcher Handler
        async function launchSystemTool(toolKey) {
            // Check MSAL token storage keys
            const token = sessionStorage.getItem("idToken") ||
                sessionStorage.getItem("accessToken") ||
                sessionStorage.getItem("vtx_token");

            if (!token || token === "true" || token.split(".").length !== 3) {
                alert("Authentication Error: Valid Entra ID JWT missing or expired. Please re-authenticate via index.html.");
                return;
            }

            try {
                const response = await fetch("/api/tools/launch", {
                    method: "POST",
                    headers: {
                        "Authorization": `Bearer ${token}`,
                        "Content-Type": "application/json"
                    },
                    body: JSON.stringify({ tool: toolKey })
                });

                const result = await response.json();

                if (response.ok) {
                    console.log(`[TOOLS] ${result.message}`);
                } else {
                    alert(`Execution Failed: ${result.message}`);
                }
            } catch (error) {
                console.error("Error launching tool:", error);
                alert("Failed to connect to local Flask agent.");
            }
        }

        async function submitNewTicket() {
            const payload = {
                clientName: document.getElementById('ingestClient').value,
                clientEmail: document.getElementById('ingestEmail').value,
                category: document.getElementById('ingestCategory').value,
                messageDetails: document.getElementById('ingestDetails').value
            };

            if (!payload.clientName || !payload.messageDetails) {
                alert('Please fill out Client Name and Message Details.');
                return;
            }

            try {
                const response = await fetch('/api/tickets', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                const result = await response.json();
                if (response.ok) {
                    alert(`Ticket ingested successfully! Serial: ${result.ticket_serial}`);
                    closeNewTicketModal();
                    loadDashboardTickets();
                } else {
                    alert(`Ingestion failed: ${result.message}`);
                }

                console.error('[INGEST ERROR]', err);
            }
